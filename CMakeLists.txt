cmake_minimum_required(VERSION 3.22)
project(GSeriesLab VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT DEFINED JUCE_DIR)
  message(FATAL_ERROR "Pass -DJUCE_DIR=/path/to/JUCE")
endif()
add_subdirectory(${JUCE_DIR} JUCE)
juce_add_plugin(GSeriesLab COMPANY_NAME "Independent DSP Lab" BUNDLE_ID "com.independentdsplab.gserieslab" PLUGIN_MANUFACTURER_CODE Idsp PLUGIN_CODE GsLb FORMATS AU Standalone PRODUCT_NAME "GSeries Lab" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD TRUE)
juce_generate_juce_header(GSeriesLab)
target_sources(GSeriesLab PRIVATE Source/PluginProcessor.cpp Source/PluginProcessor.h)
target_compile_definitions(GSeriesLab PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(GSeriesLab PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_audio_plugin_client)
